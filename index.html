ESP32 & IoT

Trayecto de Robótica: Sensores y Actuadores

Guía práctica interactiva de laboratorio y programación en protoboard

⚠️ Regla crítica de laboratorio: El ESP32 opera a 3.3V. Ningún pin GPIO tolera 5V directos. Todo sensor alimentado a 5V con línea de retorno (como el pin ECHO del HC-SR04) requiere divisor de tensión con resistencias para ingresar a 3.3V.

Nivel 1: Enclavamiento Digital (Pulsador + Módulo Relay)

Materiales del Puesto

Esquema de Conexión en Protoboard

PULSADOR:
- Terminal A ----> GPIO 18 (Usa resistencia interna INPUT_PULLUP)
- Terminal B ----> GND

MÓDULO RELAY:
- VCC        ----> VIN (5V del cable USB)
- GND        ----> GND del ESP32
- IN         ----> GPIO 23
Pegar aquí la captura del conexionado en Wokwi / Simulador

Código Fuente (C++)

#define PIN_PULSADOR 18
#define PIN_RELAY    23

bool estadoRelay = false;

void setup() {
  Serial.begin(115200);
  pinMode(PIN_PULSADOR, INPUT_PULLUP);
  pinMode(PIN_RELAY, OUTPUT);
  digitalWrite(PIN_RELAY, HIGH); // Apagado en lógica Active-LOW
  Serial.println("Relay inicializado.");
}

void loop() {
  if (digitalRead(PIN_PULSADOR) == LOW) {
    estadoRelay = !estadoRelay;
    digitalWrite(PIN_RELAY, estadoRelay ? LOW : HIGH);
    Serial.println(estadoRelay ? "RELAY: ON" : "RELAY: OFF");

    while (digitalRead(PIN_PULSADOR) == LOW) { delay(10); }
    delay(50); // Filtro antirrebote
  }
}

Nivel 2: Control Proporcional (Potenciómetro + Servomotor SG90)

Materiales del Puesto

Esquema de Conexión en Protoboard

POTENCIÓMETRO:
- Lateral 1    ----> GND
- Pin Central  ----> GPIO 34 (ADC1: rango 0 a 4095)
- Lateral 2    ----> 3.3V (¡Nunca a 5V!)

SERVOMOTOR SG90:
- Marrón (GND) ----> GND
- Rojo (VCC)   ----> VIN (5V USB)
- Naranja (PWM)----> GPIO 21
- Capacitor    ----> 470 µF en paralelo con alimentación del servo
Pegar aquí la captura del conexionado en Wokwi / Simulador

Código Fuente (C++)

#include <ESP32Servo.h>

#define PIN_POT   34
#define PIN_SERVO 21

Servo miServo;

void setup() {
  Serial.begin(115200);
  miServo.setPeriodHertz(50);
  miServo.attach(PIN_SERVO, 500, 2400);
}

void loop() {
  int valorADC = analogRead(PIN_POT);
  int angulo = map(valorADC, 0, 4095, 0, 180);
  miServo.write(angulo);

  Serial.printf("ADC: %4d -> Ángulo: %3d°\n", valorADC, angulo);
  delay(20);
}

Nivel 3: Radar de Proximidad (HC-SR04 + Divisor + Buzzer)

Materiales del Puesto

Esquema de Conexión y Divisor Resistivo

SENSOR HC-SR04:
- VCC  ----> VIN (5V USB)
- GND  ----> GND
- TRIG ----> GPIO 5

DIVISOR RESISTIVO PARA PROTEGER EL PIN ECHO:
ECHO (5V) ----[ 1k ohm ]----+----> GPIO 18 (Recibe 3.33V seguro)
                        |
                   [ 2k ohm ]
                        |
                       GND

BUZZER:
- Positivo (+) ----> GPIO 19
- Negativo (-) ----> GND
Pegar aquí la captura del conexionado en Wokwi / Simulador

Código Fuente (C++)

#define PIN_TRIG   5
#define PIN_ECHO   18
#define PIN_BUZZER 19

const int DISTANCIA_UMBRAL = 15; // cm

void setup() {
  Serial.begin(115200);
  pinMode(PIN_TRIG, OUTPUT);
  pinMode(PIN_ECHO, INPUT);
  pinMode(PIN_BUZZER, OUTPUT);
}

void loop() {
  digitalWrite(PIN_TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(PIN_TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(PIN_TRIG, LOW);

  long duracion = pulseIn(PIN_ECHO, HIGH, 30000);
  float distancia = (duracion * 0.0343) / 2.0;

  if (distancia > 0 && distancia <= DISTANCIA_UMBRAL) {
    digitalWrite(PIN_BUZZER, HIGH);
  } else {
    digitalWrite(PIN_BUZZER, LOW);
  }

  Serial.printf("Distancia: %.1f cm\n", distancia);
  delay(100);
}

Nivel 4: Consola de Control (Joystick 2 Ejes + Servo + Gatillo)

Materiales del Puesto

Esquema de Conexión en Protoboard

JOYSTICK:
- GND  ----> GND
- +5V  ----> 3.3V del ESP32 (¡Para no sobrepasar el rango del ADC!)
- VRx  ----> GPIO 34 (ADC1)
- VRy  ----> GPIO 35 (ADC1)
- SW   ----> GPIO 22 (INPUT_PULLUP)

SERVO SG90:
- VCC (Rojo)   ----> VIN (5V USB)
- GND (Marrón) ----> GND
- Señal        ----> GPIO 21

BUZZER:
- Positivo (+) ----> GPIO 19
- Negativo (-) ----> GND
Pegar aquí la captura del conexionado en Wokwi / Simulador

Código Fuente (C++)

#include <ESP32Servo.h>

#define PIN_JOY_X   34
#define PIN_JOY_Y   35
#define PIN_JOY_SW  22
#define PIN_SERVO   21
#define PIN_BUZZER  19

Servo miServo;

void setup() {
  Serial.begin(115200);
  pinMode(PIN_JOY_SW, INPUT_PULLUP);
  pinMode(PIN_BUZZER, OUTPUT);
  miServo.setPeriodHertz(50);
  miServo.attach(PIN_SERVO, 500, 2400);
}

void loop() {
  int x = analogRead(PIN_JOY_X);
  int y = analogRead(PIN_JOY_Y);
  bool disparador = (digitalRead(PIN_JOY_SW) == LOW);

  int angulo = map(x, 0, 4095, 0, 180);
  miServo.write(angulo);

  digitalWrite(PIN_BUZZER, disparador ? HIGH : LOW);

  Serial.printf("X: %4d -> %3d° | Gatillo: %d\n", x, angulo, disparador);
  delay(30);
}
